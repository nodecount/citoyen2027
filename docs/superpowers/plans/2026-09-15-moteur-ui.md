# Moteur de calcul et UI — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construire le moteur de calcul (profil + proposition → impact estimé) et les 3 écrans de l'UI (accueil/récap des thèmes, formulaire de profil, résultats), en s'appuyant sur les 12 propositions réelles déjà collectées pour Marine Le Pen, Jean-Luc Mélenchon et Édouard Philippe.

**Architecture:** Toujours une app React/Vite/TypeScript 100% statique, aucun backend. Le moteur est un ensemble de fonctions TypeScript pures et explicites (pas de moteur de règles générique) : un comportement par défaut (universel, qualitatif) et une poignée de règles d'éligibilité affinée par id de proposition. L'UI est un état local React à 3 écrans, pas de routeur.

**Tech Stack:** React 18 + TypeScript + Vite, Vitest pour les tests du moteur (pas de tests automatisés de composants React — décision actée dans la spec, revue manuelle à la place).

**Spec:** `docs/superpowers/specs/2026-09-15-moteur-ui-design.md`

## Global Constraints

- Stack : React (Vite) + TypeScript, application 100% statique, aucun backend.
- Neutralité : candidats toujours en ordre alphabétique fixe (Le Pen, Mélenchon, Philippe) — jamais par score.
- Comportement de calcul par défaut : `concerne: true`, `niveau: "qualitatif"`, jamais de montant par ménage inventé à partir d'un chiffrage macro — un chiffrage macro disponible est mentionné tel quel, jamais attribué au foyer de l'utilisateur.
- Seules les propositions listées dans `src/engine/regles.ts` (voir Task 3) ont un comportement d'éligibilité affiné ; toutes les autres utilisent le comportement par défaut.
- Pour un thème sans proposition `"complet"` pour un candidat donné, afficher explicitement « Programme en cours d'analyse sur ce thème » — jamais masquer la section.
- Pas de tests automatisés de composants React dans ce plan (décision actée) — seul le moteur (`src/engine/`) a des tests Vitest. Revue manuelle via `npm run dev` en fin de plan.
- Pas de nouvelle collecte de données, pas de déploiement, pas de travail de direction artistique poussé (hors scope, spec).

import type { Profil, Proposition, Impact } from "../types";

export function evaluerParDefaut(_profil: Profil, proposition: Proposition): Impact {
  let explication = proposition.resume_factuel;
  if (proposition.chiffrage_disponible) {
    explication +=
      " Chiffrage macro (coût ou gain estimé au niveau national, pas un montant par ménage) : " +
      proposition.hypotheses_calcul;
  }
  return {
    concerne: true,
    niveau: "qualitatif",
    explication,
  };
}

```ts
import type { Profil, Proposition, Impact } from "../types";
import { evaluerParDefaut } from "./evaluerParDefaut";

function avecConcerneAffine(
  profil: Profil,
  proposition: Proposition,
  concerne: boolean,
  noteEligibilite: string
): Impact {
  const base = evaluerParDefaut(profil, proposition);
  const explication = concerne
    ? `${noteEligibilite} ${base.explication}`
    : `${noteEligibilite} Cette mesure ne s'applique donc pas à votre profil tel que renseigné.`;
  return { ...base, concerne, explication };
}

export const regles: Record<string, (profil: Profil, proposition: Proposition) => Impact> = {
  "le-pen-fisc-002": (profil, proposition) => {
    const concerne = profil.tranche_age === "moins_30";
    return avecConcerneAffine(
      profil,
      proposition,
      concerne,
      concerne
        ? "Vous avez moins de 30 ans : cette mesure vous concerne directement."
        : "Cette mesure cible les moins de 30 ans."
    );
  },

  "le-pen-pa-002": (profil, proposition) => {
    const concerne = profil.statut_professionnel === "retraite";
    return avecConcerneAffine(
      profil,
      proposition,
      concerne,
      concerne
        ? "Vous êtes retraité : cette mesure vous concerne directement."
        : "Cette mesure concerne les retraités."
    );
  },

  "melenchon-fisc-002": (profil, proposition) => {
    const concerne = profil.tranche_patrimoine === "500k_1_3m" || profil.tranche_patrimoine === "plus_1_3m";
    const nuance =
      profil.tranche_patrimoine === "500k_1_3m"
        ? "Votre tranche de patrimoine peut approcher le seuil d'entrée de l'ISF proposé (1,3 million d'euros) selon votre situation exacte."
        : "Votre patrimoine dépasse le seuil d'entrée de l'ISF proposé (1,3 million d'euros).";
    return avecConcerneAffine(
      profil,
      proposition,
      concerne,
      concerne ? nuance : "Cette mesure ne concerne que les patrimoines dépassant 1,3 million d'euros."
    );
  },

  "melenchon-pa-001": (profil, proposition) => {
    const concerne =
      profil.statut_professionnel === "salarie" &&
      (profil.tranche_revenus === "moins_1200" || profil.tranche_revenus === "1200_2000");
    return avecConcerneAffine(
      profil,
      proposition,
      concerne,
      concerne
        ? "Vous êtes salarié avec un revenu proche du SMIC actuel : cette mesure vous concerne directement."
        : "Cette mesure concerne les salariés rémunérés au SMIC ou à un niveau proche."
    );
  },

  "philippe-pa-001": (profil, proposition) => {
    const concerne = ["salarie", "independant", "fonctionnaire"].includes(profil.statut_professionnel);
    return avecConcerneAffine(
      profil,
      proposition,
      concerne,
      concerne
        ? "Vous êtes en activité : cette mesure vous concerne directement."
        : "Cette mesure concerne les actifs (salariés, indépendants, fonctionnaires)."
    );
  },
};
```

export function evaluerProposition(profil: Profil, proposition: Proposition): Impact {

```tsx
import { useState, type FormEvent } from "react";
import profilSchema from "../../data/profils-schema.json";
import type { ChampProfil, Profil } from "../types";

const champs: ChampProfil[] = profilSchema as ChampProfil[];

interface ProfilFormProps {
  onValider: (profil: Profil) => void;
}

type ValeursFormulaire = Record<string, string>;

function valeurParDefaut(champ: ChampProfil): string {
  if (champ.type === "enum" && champ.options && champ.options.length > 0) {
    return champ.options[0];
  }
  if (champ.type === "number") {
    return "0";
  }
  return "";
}

export default function ProfilForm({ onValider }: ProfilFormProps) {
  const [valeurs, setValeurs] = useState<ValeursFormulaire>(() => {
    const initiales: ValeursFormulaire = {};
    for (const champ of champs) {
      initiales[champ.id] = valeurParDefaut(champ);
    }
    return initiales;
  });

  function handleChange(id: string, valeur: string) {
    setValeurs((precedent) => ({ ...precedent, [id]: valeur }));
  }

  function handleSubmit(event: FormEvent) {
    event.preventDefault();
    const profil = {
      statut_professionnel: valeurs.statut_professionnel,
      tranche_revenus: valeurs.tranche_revenus,
      situation_familiale: valeurs.situation_familiale,
      nombre_enfants: Number(valeurs.nombre_enfants) || 0,
      logement: valeurs.logement,
      secteur_activite: valeurs.secteur_activite,
      region: valeurs.region,
      tranche_age: valeurs.tranche_age,
      tranche_patrimoine: valeurs.tranche_patrimoine,
    } satisfies Profil;
    onValider(profil);
  }

  return (
    <form onSubmit={handleSubmit}>
      <h1>Votre profil</h1>
      {champs.map((champ) => (
        <div key={champ.id}>
          <label htmlFor={champ.id}>{champ.label}</label>
          {champ.type === "enum" ? (
            <select
              id={champ.id}
              value={valeurs[champ.id]}
              onChange={(e) => handleChange(champ.id, e.target.value)}
              required
            >
              {(champ.options ?? []).map((option) => (
                <option key={option} value={option}>
                  {option}
                </option>
              ))}
            </select>
          ) : champ.type === "number" ? (
            <input
              id={champ.id}
              type="number"
              min={0}
              value={valeurs[champ.id]}
              onChange={(e) => handleChange(champ.id, e.target.value)}
              required
            />
          ) : (
            <input
              id={champ.id}
              type="text"
              value={valeurs[champ.id]}
              onChange={(e) => handleChange(champ.id, e.target.value)}
              required
            />
          )}
        </div>
      ))}
      <button type="submit">Voir les résultats</button>
    </form>
  );
}
```

import themes from "../../data/themes.json";
import lePenMeta from "../../data/programmes/le-pen/meta.json";

const themesTypes: Theme[] = themes as Theme[];

const candidats: { meta: CandidatMeta; propositions: Proposition[] }[] = [
  { meta: lePenMeta as CandidatMeta, propositions: lePenProps as Proposition[] },
  { meta: melenchonMeta as CandidatMeta, propositions: melenchonProps as Proposition[] },
  { meta: philippeMeta as CandidatMeta, propositions: philippeProps as Proposition[] },
].sort((a, b) => a.meta.slug.localeCompare(b.meta.slug));

  profil: Profil;
}

  return (
    <section>
      <h1>Résultats</h1>
      {candidats.map(({ meta, propositions }) => (
        <article key={meta.slug}>
          <h2>{meta.nom}</h2>
          {themesTypes.map((theme) => {
            const propsCompletesDuTheme = propositions.filter(
              (p) => p.theme === theme.id && p.statut === "complet"

            return (

                ) : (

                              <em>Cette mesure ne s'applique pas à votre profil.</em>
                            </p>
                          )}
                        </li>

                )}
              </div>
            );
          })}
        </article>
      ))}

  );
}

- [ ] **Step 2: Réécrire `src/App.tsx`**

```tsx
import { useState } from "react";
import Accueil from "./ui/Accueil";
import ProfilForm from "./ui/ProfilForm";
import Resultats from "./ui/Resultats";
import type { Profil } from "./types";

type Ecran = "accueil" | "formulaire" | "resultats";

export default function App() {
  const [ecran, setEcran] = useState<Ecran>("accueil");
  const [profil, setProfil] = useState<Profil | null>(null);

  if (ecran === "accueil") {
    return <Accueil onCommencer={() => setEcran("formulaire")} />;
  }

  if (ecran === "formulaire") {
    return (
      <ProfilForm
        onValider={(p) => {
          setProfil(p);
          setEcran("resultats");
        }}
      />
    );
  }

  if (profil) {
    return <Resultats profil={profil} />;
  }

  return <Accueil onCommencer={() => setEcran("formulaire")} />;
}
```
